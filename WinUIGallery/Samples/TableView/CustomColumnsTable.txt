--- c#
public ObservableCollection<TableViewSampleItem> CustomColumnItems { get; } =
[
    new(1, "Item 1", "Alpha", 25),
    new(2, "Item 2", "Beta", 80),
    new(3, "Item 3", "Gamma", 50),
    new(4, "Item 4", "Alpha", 100),
    new(5, "Item 5", "Beta", 10),
];

public sealed class TableViewSampleItem(int id, string name, string category, int value)
{
    public int Id { get; } = id;
    public string Name { get; set; } = name;
    public string Category { get; set; } = category;
    public int Value { get; } = value;
    public string Status => Id % 2 == 0 ? "Paused" : "Active";
    public int Rating => (Id - 1) % 5 + 1;
}
