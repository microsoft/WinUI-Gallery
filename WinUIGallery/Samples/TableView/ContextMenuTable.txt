--- header
Adding context menus to cells, rows, and headers
--- xaml
<tabular:TableView
    x:Name="ContextMenuTable"
    Height="280"
    MaxWidth="480"
    HorizontalAlignment="Left"
    AutomationProperties.Name="Items with context menus"
    CanUserResizeColumns="False"
    CanUserSortColumns="False"
    IsReadOnly="True"
    ItemsSource="{x:Bind ContextMenuItems}"
    SelectionMode="Single">
    <tabular:TableView.Resources>
        <!-- Menu items inherit the requested row's item as their DataContext. -->
        <MenuFlyout x:Key="NameCellMenu">
            <MenuFlyoutItem Click="CopyName_Click" Text="Copy name">
                <MenuFlyoutItem.Icon>
                    <SymbolIcon Symbol="Copy" />
                </MenuFlyoutItem.Icon>
            </MenuFlyoutItem>
        </MenuFlyout>
        <MenuFlyout x:Key="RowMenu">
            <MenuFlyoutItem Click="CopyRow_Click" Text="Copy row">
                <MenuFlyoutItem.Icon>
                    <SymbolIcon Symbol="Copy" />
                </MenuFlyoutItem.Icon>
            </MenuFlyoutItem>
            <MenuFlyoutItem Click="NextCategory_Click" Text="Move to next category" />
            <MenuFlyoutSeparator />
            <MenuFlyoutItem Click="RemoveRow_Click" Text="Remove">
                <MenuFlyoutItem.Icon>
                    <SymbolIcon Symbol="Delete" />
                </MenuFlyoutItem.Icon>
            </MenuFlyoutItem>
        </MenuFlyout>
        <!-- Rows get their menu from an implicit style. -->
        <Style TargetType="tabular:TableViewRow">
            <Setter Property="ContextFlyout" Value="{StaticResource RowMenu}" />
        </Style>
    </tabular:TableView.Resources>
    <!-- The table's own menu opens when no cell, row, or header menu applies. -->
    <tabular:TableView.ContextFlyout>
        <MenuFlyout>
            <MenuFlyoutItem Click="RestoreItems_Click" Text="Restore all items" />
        </MenuFlyout>
    </tabular:TableView.ContextFlyout>
    <tabular:TableView.Columns>
        <tabular:TableViewTextColumn
            Binding="{Binding Name}"
            CellContextFlyout="{StaticResource NameCellMenu}"
            Header="Name"
            Width="*">
            <tabular:TableViewTextColumn.HeaderContextFlyout>
                <MenuFlyout Opening="HeaderMenu_Opening">
                    <MenuFlyoutItem Click="HideColumn_Click" Text="Hide column" />
                    <MenuFlyoutItem Click="ShowAllColumns_Click" Text="Show all columns" />
                </MenuFlyout>
            </tabular:TableViewTextColumn.HeaderContextFlyout>
        </tabular:TableViewTextColumn>
        <tabular:TableViewTextColumn
            Binding="{Binding Category}"
            Header="Category"
            Width="140">
            <tabular:TableViewTextColumn.HeaderContextFlyout>
                <MenuFlyout Opening="HeaderMenu_Opening">
                    <MenuFlyoutItem Click="HideColumn_Click" Text="Hide column" />
                    <MenuFlyoutItem Click="ShowAllColumns_Click" Text="Show all columns" />
                </MenuFlyout>
            </tabular:TableViewTextColumn.HeaderContextFlyout>
        </tabular:TableViewTextColumn>
        <tabular:TableViewTextColumn
            Binding="{Binding Value}"
            Header="Value"
            Width="100">
            <tabular:TableViewTextColumn.HeaderContextFlyout>
                <MenuFlyout Opening="HeaderMenu_Opening">
                    <MenuFlyoutItem Click="HideColumn_Click" Text="Hide column" />
                    <MenuFlyoutItem Click="ShowAllColumns_Click" Text="Show all columns" />
                </MenuFlyout>
            </tabular:TableViewTextColumn.HeaderContextFlyout>
        </tabular:TableViewTextColumn>
    </tabular:TableView.Columns>
</tabular:TableView>
<TextBlock
    x:Name="ContextMenuStatusText"
    AutomationProperties.LiveSetting="Polite"
    Text="No menu command chosen."
    TextWrapping="Wrap" />
--- c#
public ObservableCollection<TableViewSampleItem> ContextMenuItems { get; } = TableViewSampleItem.CreateItems();

private TableViewColumn? _headerMenuColumn;

private void CopyName_Click(object sender, RoutedEventArgs e)
{
    if (sender is FrameworkElement { DataContext: TableViewSampleItem item })
    {
        CopyToClipboard(item.Name);
    }
}

private void CopyRow_Click(object sender, RoutedEventArgs e)
{
    if (sender is FrameworkElement { DataContext: TableViewSampleItem item })
    {
        CopyToClipboard($"{item.Id}\t{item.Name}\t{item.Category}\t{item.Value}");
    }
}

private void NextCategory_Click(object sender, RoutedEventArgs e)
{
    if (sender is FrameworkElement { DataContext: TableViewSampleItem item })
    {
        string[] categories = ["Alpha", "Beta", "Gamma"];
        item.Category = categories[(Array.IndexOf(categories, item.Category) + 1) % categories.Length];
        ContextMenuStatusText.Text = $"Moved {item.Name} to {item.Category}.";
    }
}

private void RemoveRow_Click(object sender, RoutedEventArgs e)
{
    if (sender is FrameworkElement { DataContext: TableViewSampleItem item } && ContextMenuItems.Remove(item))
    {
        ContextMenuStatusText.Text = $"Removed {item.Name}.";
    }
}

private void RestoreItems_Click(object sender, RoutedEventArgs e)
{
    ContextMenuItems.Clear();
    foreach (TableViewSampleItem item in TableViewSampleItem.CreateItems())
    {
        ContextMenuItems.Add(item);
    }

    ContextMenuStatusText.Text = "Restored all items.";
}

private void HeaderMenu_Opening(object sender, object e)
{
    MenuFlyout menu = (MenuFlyout)sender;
    _headerMenuColumn = ContextMenuTable.Columns.First(column => column.HeaderContextFlyout == menu);
    if (menu.Items[0] is MenuFlyoutItem hideItem)
    {
        hideItem.Text = $"Hide {_headerMenuColumn?.Header}";
        hideItem.IsEnabled = _headerMenuColumn is not null &&
            ContextMenuTable.Columns.Count(column => column.Visibility == Visibility.Visible) > 1;
    }
}

private void HideColumn_Click(object sender, RoutedEventArgs e)
{
    if (_headerMenuColumn is not null)
    {
        _headerMenuColumn.Visibility = Visibility.Collapsed;
        ContextMenuStatusText.Text = $"Hid the {_headerMenuColumn.Header} column.";
    }
}

private void ShowAllColumns_Click(object sender, RoutedEventArgs e)
{
    foreach (TableViewColumn column in ContextMenuTable.Columns)
    {
        column.Visibility = Visibility.Visible;
    }

    ContextMenuStatusText.Text = "Showed all columns.";
}

private void CopyToClipboard(string text)
{
    DataPackage package = new();
    package.SetText(text);
    Clipboard.SetContent(package);
    ContextMenuStatusText.Text = $"Copied \"{text}\".";
}

// TableViewSampleItem is the item class from the "A basic TableView" example. Its Name and Category
// setters raise PropertyChanged, so "Move to next category" updates the cell.
