public ObservableCollection<TableViewSampleItem> PresentationItems { get; } = CreateItems();

private void EmptyStateToggle_Toggled(object sender, RoutedEventArgs e)
{
    if (EmptyStateToggle.IsOn)
    {
        PresentationItems.Clear();
    }
    else if (PresentationItems.Count == 0)
    {
        foreach (TableViewSampleItem item in CreateItems())
        {
            PresentationItems.Add(item);
        }
    }
}

private static ObservableCollection<TableViewSampleItem> CreateItems()
{
    string[] categories = ["Alpha", "Beta", "Gamma"];
    int[] values = [25, 80, 50, 100, 10];
    ObservableCollection<TableViewSampleItem> items = [];

    for (int index = 0; index < 1000; index++)
    {
        items.Add(new TableViewSampleItem(
            index + 1,
            $"Item {index + 1}",
            categories[index % categories.Length],
            values[index % values.Length]));
    }

    return items;
}
