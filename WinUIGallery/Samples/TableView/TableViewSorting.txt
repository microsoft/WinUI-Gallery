--- header
Sort rows by selecting a column header
--- xaml
<tabular:TableView
    x:Name="SortingTable"
    Height="300"
    CanUserSortColumns="True"
    ItemsSource="{x:Bind Employees}"
    Sorted="SortingTable_Sorted">
    <tabular:TableViewTextColumn Header="Name" Binding="{Binding Name}" Width="*" />
    <tabular:TableViewTextColumn Header="Department" Binding="{Binding Department}" Width="*" />
    <tabular:TableViewTextColumn Header="Location" Binding="{Binding Location}" Width="*" />
    <!-- Metric columns usually open on the largest values. -->
    <tabular:TableViewTextColumn
        Header="Performance"
        Binding="{Binding Performance}"
        SortCycle="DescendingAscendingNone"
        Width="130" />
    <!-- A column can opt out of sorting. -->
    <tabular:TableViewTextColumn Header="Notes" Binding="{Binding Notes}" CanSort="False" Width="*" />
</tabular:TableView>
--- c#
private void SortingTable_Sorted(TableView sender, TableViewSortedEventArgs args)
{
    SortingOutput.Text = args.Direction == SortDirection.None
        ? "Rows are in source order."
        : $"Sorted by {args.Column.Header}, {args.Direction}.";
}

private void ClearSort_Click(object sender, RoutedEventArgs e)
{
    SortingTable.ClearSort();
}
