--- header
LayoutPanel lets a custom Layout arrange ordinary panel children.
--- xaml
<LayoutPanel
    BorderBrush="{ThemeResource CardStrokeColorDefaultBrush}"
    BorderThickness="1"
    CornerRadius="8"
    Padding="12">
    <LayoutPanel.Layout>
        <layouts:FeaturedTileLayout Spacing="8"/>
    </LayoutPanel.Layout>

    <Border Height="96">
        <TextBlock Text="Featured item"/>
    </Border>
    <Border Height="72">
        <TextBlock Text="Second"/>
    </Border>
    <Border Height="92">
        <TextBlock Text="Third"/>
    </Border>
    <Border Height="80">
