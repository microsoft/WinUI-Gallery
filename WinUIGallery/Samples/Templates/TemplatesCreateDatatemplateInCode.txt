--- header
Create a DataTemplate in code (experimental)
--- xaml
<ComboBox x:Name="CodeTemplateComboBox" Header="Options" SelectedIndex="0">
    <ComboBox.Items>
        <x:String>Option 1</x:String>
        <x:String>Option 2</x:String>
        <x:String>Option 3</x:String>
    </ComboBox.Items>
</ComboBox>
--- c#
// The factory runs once for every item container that uses the template.
CodeTemplateComboBox.ItemTemplate = new DataTemplate(() =>
{
    FontIcon icon = new FontIcon { Glyph = "\uE734", FontSize = 14 };

    TextBlock text = new TextBlock();
    text.SetBinding(TextBlock.TextProperty, new Binding());

    StackPanel panel = new StackPanel { Orientation = Orientation.Horizontal, Spacing = 8 };
    panel.Children.Add(icon);
    panel.Children.Add(text);
    return panel;
});
