        sometimes that is not easy to do.
    -->
    <TextBlock Grid.Row="1" VerticalAlignment="Center" Text="Row 1"/>
    <Button Grid.Column="1" Grid.Row="1" Content="First stop" TabIndex="1" HorizontalAlignment="Stretch"/>
    <Button Grid.Column="2" Grid.Row="1" Content="Third stop" TabIndex="3" HorizontalAlignment="Stretch"/>

    <TextBlock Grid.Row="2" VerticalAlignment="Center" Text="Row 2"/>
    <Button Grid.Column="1" Grid.Row="2" Content="Second stop" TabIndex="2" HorizontalAlignment="Stretch"/>
    <Button Grid.Column="2" Grid.Row="2" Content="Not a stop" IsTabStop="False" HorizontalAlignment="Stretch"/>
