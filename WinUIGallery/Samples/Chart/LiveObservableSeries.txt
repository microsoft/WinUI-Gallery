--- c#
private double[] _responseTimes = [184, 179, 173, 171, 168];

public double[] ResponseTimes
{
    get => _responseTimes;
    private set
    {
        _responseTimes = value;
        PropertyChanged?.Invoke(this, new PropertyChangedEventArgs(nameof(ResponseTimes)));
    }
}

private void AddResponseTimeSampleButton_Click(object sender, RoutedEventArgs e)
{
    // Samples.ItemsSource reads its values when it is set, so assign a new
    // array instead of changing the existing one. Keep the eight most recent values.
    ResponseTimes = [.. ResponseTimes.TakeLast(7), 176];
}
