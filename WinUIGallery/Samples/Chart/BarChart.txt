--- header
Create a bar chart and sort its categories
--- xaml
<charts:Chart>
    <charts:Chart.Data>
        <charts:Samples x:Name="Region"
                        ItemsSource="{x:Bind Regions, Mode=OneTime}" />
        <charts:Samples x:Name="Units"
                        ItemsSource="{x:Bind UnitsSold, Mode=OneTime}" />
    </charts:Chart.Data>

    <charts:Chart.Axes>
        <charts:CategoryAxis x:Name="RegionX"
                             Label="Region"
                             SortKey="$(SortKey)"
                             SortOrder="$(SortOrder)" />
        <charts:LinearAxis x:Name="UnitsY"
                           Label="Units sold"
                           Minimum="0"
                           GridLines="Major" />
    </charts:Chart.Axes>

    <charts:BarSeries Title="Units sold"
                      Orientation="$(Orientation)"
                      Fill="#0078D4"
                      XAxis="{x:Bind RegionX, Mode=OneTime}"
                      YAxis="{x:Bind UnitsY, Mode=OneTime}"
                      XValues="{x:Bind Region, Mode=OneTime}"
                      YValues="{x:Bind Units, Mode=OneTime}" />
</charts:Chart>
--- c#
public string[] Regions { get; } = ["North", "South", "East", "West"];

public double[] UnitsSold { get; } = [42, 38, 51, 47];
