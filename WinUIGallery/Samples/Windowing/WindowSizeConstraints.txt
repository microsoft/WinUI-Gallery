--- header
Constrain the client size with minimum and maximum dimensions
--- xaml
<!-- SampleWindow.xaml; keep the template's InitializeComponent() constructor. -->
<Window
    x:Class="MyApp.SampleWindow"
    xmlns="http://schemas.microsoft.com/winfx/2006/xaml/presentation"
    xmlns:x="http://schemas.microsoft.com/winfx/2006/xaml"
    Width="640"
    Height="480"
    MinWidth="$(MinWidth)"
    MinHeight="$(MinHeight)"
    MaxWidth="$(MaxWidth)"
    MaxHeight="$(MaxHeight)">
    <TextBlock Margin="24" Text="Resize this window." />
</Window>
<!-- Disable maximizing in code-behind, as shown in C#. -->
--- c#
// The same size constraints can be set in C# instead of XAML.
Window window = new Window
{
    Width = 640,
    Height = 480,
    MinWidth = $(MinWidth),
    MinHeight = $(MinHeight),
    MaxWidth = $(MaxWidth),
    MaxHeight = $(MaxHeight),
    Content = new TextBlock { Margin = new Thickness(24), Text = "Resize this window." }
};

if (window.AppWindow.Presenter is OverlappedPresenter presenter)
{
    presenter.IsMaximizable = false;
}

// Gallery tracks child windows so they close when the app closes.
WindowHelper.TrackWindow(window);
// Use Window.Activate() for the first show, including for XAML-defined windows.
window.Activate();
