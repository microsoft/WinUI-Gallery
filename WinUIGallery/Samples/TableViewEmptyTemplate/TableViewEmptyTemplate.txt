--- header
Show custom content when the table has no rows
--- xaml
<tabular:TableView Height="300" ItemsSource="{x:Bind Employees}">
    <tabular:TableView.EmptyTemplate>
        <DataTemplate>
            <StackPanel HorizontalAlignment="Center" VerticalAlignment="Center" Spacing="8">
                <FontIcon FontSize="32" Glyph="&#xE716;" />
                <TextBlock Style="{ThemeResource BodyStrongTextBlockStyle}" Text="No employees to show" />
            </StackPanel>
        </DataTemplate>
    </tabular:TableView.EmptyTemplate>
    <tabular:TableViewTextColumn Header="Name" Binding="{Binding Name}" Width="*" />
    <tabular:TableViewTextColumn Header="Role" Binding="{Binding Role}" Width="*" />
    <tabular:TableViewTextColumn Header="Department" Binding="{Binding Department}" Width="*" />
    <tabular:TableViewTextColumn Header="Location" Binding="{Binding Location}" Width="*" />
</tabular:TableView>
--- c#
private void EmptyStateToggle_Toggled(object sender, RoutedEventArgs e)
{
    // EmptyTemplate is shown whenever the items source has no items.
    Employees.Clear();
}
